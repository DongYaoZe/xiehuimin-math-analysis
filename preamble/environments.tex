\newcommand{\xieEnvTitle}[2]{%
  \noindent\textcolor{#1}{\sffamily\bfseries #2}\quad\ignorespaces
}
\newcommand{\xieNumberedTitle}[3]{%
  \IfValueTF{#3}
    {\xieEnvTitle{#1}{#2~#3}}
    {\xieEnvTitle{#1}{#2}}%
}

\NewDocumentEnvironment{xdefinition}{o}
  {\begin{tcolorbox}[xie/theorem]\xieNumberedTitle{xieNavy}{定义}{#1}}
  {\end{tcolorbox}}
