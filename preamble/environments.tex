\newcommand{\xieEnvTitle}[2]{%
  \noindent\textcolor{#1}{\sffamily\bfseries #2}\quad\ignorespaces
}
\newcommand{\xieNumberedTitle}[3]{%
  \IfValueTF{#3}
    {\xieEnvTitle{#1}{#2~#3}}
    {\xieEnvTitle{#1}{#2}}%
}

% Exercise/solution material occurs several thousand times in the inline
% edition.  Using an enhanced breakable tcolorbox for every such unit makes a
% full-book cloud build prohibitively slow.  These flow blocks keep the same
% visual hierarchy with light rules and indentation while remaining ordinary
% page-breakable TeX paragraphs.
\newcommand{\xieFlowBegin}[3]{%
  \par\addvspace{.72em plus .15em}%
  \Needspace{4\baselineskip}%
  {\color{#1}\hrule height .55pt}%
  \nobreak\vspace{.42em}%
  \begingroup
  \setlength{\leftskip}{.65em}%
  \setlength{\rightskip}{.25em}%
  \xieNumberedTitle{#1}{#2}{#3}%
}
\newcommand{\xieFlowEnd}[1]{%
  \par\endgroup
  \vspace{.32em}%
  {\color{#1!55}\hrule height .28pt}%
  \par\addvspace{.72em plus .15em}%
}
\newcommand{\xieSolutionBegin}{%
  \par\addvspace{.54em plus .12em}%
  \Needspace{4\baselineskip}%
  {\color{xieTeal!70}\hrule height .38pt}%
  \nobreak\vspace{.36em}%
  \begingroup
  \setlength{\leftskip}{.85em}%
  \setlength{\rightskip}{.25em}%
  \xieEnvTitle{xieTeal}{解答}%
}
\newcommand{\xieSolutionEnd}{%
  \par\endgroup
  \vspace{.25em}%
  {\color{xieRule}\hrule height .25pt}%
  \par\addvspace{.6em plus .12em}%
}
